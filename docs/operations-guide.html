1快速开始

引入 encrypt

<dependency>
    <groupId>io.gitee.sander-o</groupId>
    <artifactId>encrypt</artifactId>
    <version>1.0.6.7-d01</version>
</dependency>

版本号以 Maven Central 实际发布为准。只做加密就只引 encrypt；Spring Boot 自动装配加 encrypt-spring-boot-starter（见 第 5 节）；bedrock、fortify 是可选的配套模块。

三行代码开始使用

// 摘要：MD5（大写十六进制输出）
String md5 = MD5Utils.encode("password");

// 对称加密：AES/CBC/PKCS5Padding，密钥为 Hex 字符串，IV 为 16 字节
SecretKey key = AesKeyGenerator.getSecretKey();
AesProperties props = new AesProperties();
props.setSecretKey(key);
props.setIvParameter("0123456789abcdef");
AesHelper.load(props);
String ciphertext = AesHelper.encode("hello");

4.1摘要（MD5 / SHA）

String md5 = MD5Utils.encode("password");            // 大写十六进制
String lower = MD5Utils.encode("password", true);    // 小写（BigInteger 路径，可能丢失前导零）
String sha256 = ShaUtils.encode("password");         // 默认 SHA-256，小写十六进制
String sha512 = ShaUtils.encode("SHA-512", "password");
⚠️ MD5 不具备抗碰撞性，仅用于数据指纹等非安全场景，请勿用于密码存储。

4.2Base64 / CRC32

String encoded = Base64Utils.encode("Hello World");  // 标准 Base64，UTF-8
String decoded = Base64Utils.decode(encoded);

long crc = Crc32Utils.encode("data".getBytes());

4.3AES 对称加解密

// 1. 生成随机密钥（也可用自定义 Hex 字符串，16/24/32 字节对应 AES-128/192/256）
SecretKey key = AesKeyGenerator.getSecretKey();
String hexKey = AesKeyGenerator.getSecretKeyStr(key); // 便于持久化的 Hex 字符串

// 2. 加载配置：CBC 模式下 IV 必须是 16 字节（UTF-8 字节数）
AesProperties props = new AesProperties();
props.setSecretKey(key);
props.setIvParameter("0123456789abcdef");
AesHelper.load(props);

// 3. 加解密（密文为 Base64）
String ciphertext = AesHelper.encode("敏感数据");
String plaintext  = AesHelper.decode(ciphertext);     // "敏感数据"

// 不依赖全局配置，指定密钥重载：
String c2 = AesHelper.encode(hexKey, "敏感数据");
String p2 = AesHelper.decode(hexKey, c2);

4.4DES 对称加解密

DesProperties props = new DesProperties();
props.setSecretKeyStr("BeYour18");                       // UTF-8 编码后必须恰好 8 字节
props.setSecretKey(DesKeyGenerator.getSecretKey("DES", "BeYour18"));
// props.setAlgorithmMode("DES/ECB/NoPadding");         // 可选模式
DesHelper.load(props);

// 只提供字符串密钥时，load 会自动按 UTF-8 原文构造密钥对象：
DesProperties props2 = new DesProperties();
props2.setSecretKeyStr("12345678");
DesHelper.load(props2);

String ciphertext = DesHelper.encode("敏感数据");
String plaintext  = DesHelper.decode(ciphertext);
⚠️ DES 默认模式即 ECB，IV 不生效。为保证既有密文可解默认值未改动；建议优先使用 AES，或在配置中显式指定 IV 生效的模式。

4.5IDEA 对称加解密

// 密钥为 Hex 字符串（32 个 Hex 字符 = 128 位），依赖 BouncyCastle Provider（自动注册）
IdeaProperties props = new IdeaProperties();
props.setSecretKeyStr("b1d58f20dfe96a6659ad818370500e66");
props.setSecretKey(IdeaKeyGenerator.getSecretKey("IDEA", "b1d58f20dfe96a6659ad818370500e66"));
IdeaHelper.load(props);

String ciphertext = IdeaHelper.encode("敏感数据");
String plaintext  = IdeaHelper.decode(ciphertext);

// 显式传参、不依赖全局配置：
SecretKey key = new SecretKeySpec("0123456789abcdef".getBytes(StandardCharsets.UTF_8), "IDEA");
String c2 = IdeaUtils.encode("IDEA", "IDEA/ECB/ISO10126Padding", key, "敏感数据");
String p2 = IdeaUtils.decode("IDEA", "IDEA/ECB/ISO10126Padding", key, c2);

4.6RSA 加解密与签名

// 1. 生成密钥对（默认 2048 位），Base64(X.509) 公钥 / Base64(PKCS#8) 私钥
RsaKeyGenerator.getKeyPair();
String publicKey = RsaKeyGenerator.getPublicKeyStr();
String privateKey = RsaKeyGenerator.getPrivateKeyStr();

// 2. 加载配置（默认 OAEPWithSHA-256AndMGF1Padding，签名默认 SHA256withRSA）
RsaProperties props = new RsaProperties();
props.setPublicKeyStr(publicKey);
props.setPrivateKeyStr(privateKey);
props.setPublicKey(RsaKeyGenerator.getPublicKey(publicKey));
props.setPrivateKey(RsaKeyGenerator.getPrivateKey(privateKey));
RsaHelper.load(props);

// 3. 加解密（超长明文自动分段）
String ciphertext = RsaHelper.encode("敏感数据");
String plaintext  = RsaHelper.decode(ciphertext);

// 4. 签名 / 验签
String sign = RsaHelper.sign("待签名内容");
boolean pass = RsaHelper.verify("待签名内容", sign);
⚠️ RsaProperties 修改 keyBits / algorithmMode 后需调用 recalcMaxBlocks() 同步分块大小。

4.7DSA 签名

java.security.KeyPair keyPair = DsaKeyGenerator.getKeyPair();   // 默认 SHA256withDSA，1024 位
DsaProperties props = new DsaProperties();
props.setSignAlgorithm("SHA256withDSA");
props.setPublicKey(keyPair.getPublic());
props.setPrivateKey(keyPair.getPrivate());
DsaHelper.load(props);

String sign = DsaHelper.sign("待签名内容");
boolean pass = DsaHelper.verify("待签名内容", sign);

4.8JWT 签发与解析

Map<String, String> payload = new HashMap<>();
payload.put("userId", "123");

// 始终显式传入签名密钥；类内置默认值仅供演示
String token = JWTUtils.generatorToken(payload, "your-strong-secret");
Map<String, String> parsed = JWTUtils.parseToken(token, "your-strong-secret");

// 带过期时间（签名 / 过期校验失败抛 JWTVerificationException）
String expiring = JWTUtils.generatorToken(payload,
        new Date(System.currentTimeMillis() + 3600_000L), "your-strong-secret");

5Spring Boot 5 分钟接入

第 1 步：引入依赖

<dependency>
    <groupId>io.gitee.sander-o.spring.boot</groupId>
    <artifactId>encrypt-spring-boot-starter</artifactId>
    <version>1.0.6.7-d01</version>
</dependency>

第 2 步：启用自动装配

@SpringBootApplication
@EnableToolsEncrypt
public class Application {
    public static void main(String[] args) {
        SpringApplication.run(Application.class, args);
    }
}

第 3 步：配置 application.yml

tools:
  encrypt:
    symmetrical:
      aes:
        enabled: true                          # 是否启用 AES 自动装配
        algorithm-mode: AES/CBC/PKCS5Padding   # 模式/填充
        iv-parameter: "0123456789abcdef"       # CBC 模式必须为 16 字节（UTF-8）
        secret-key: "e5f7a1c93b8d24670fa1c93b8d24670f"   # Hex 字符串
        group-enabled: true                    # 多密钥分组（按 name 切换）
        group:
          - name: phone
            secret-key: "a1b2c3d4e5f60718293a4b5c6d7e8f90"
            iv-parameter: "abcdef0123456789"
      des:
        enabled: true
        secret-key: "BeYour18"                 # UTF-8 后必须恰好 8 字节
      idea:
        enabled: false
    asymmetric:
      rsa:
        enabled: true
        algorithm-mode: RSA/ECB/OAEPWITHSHA-256ANDMGF1PADDING
        sign-algorithm: SHA256withRSA          # JDK 8 默认 Provider 不支持 SHA3-256withRSA
        key-bits: 2048
        public-key: "MIGfMA0GCSqGSIb3DQEBAQUAA4GN...（Base64 X.509 公钥）"
        private-key: "MIICdgIBADANBgkqhkiG9w0BAQ...（Base64 PKCS#8 私钥）"

第 4 步：业务代码直接调用

@RestController
@RequestMapping("/api/encrypt")
public class EncryptController {

    @GetMapping("/aes/encode")
    public String aesEncode(@RequestParam String data) throws AesException {
        return AesHelper.encode(data);
    }

    @GetMapping("/aes/decode")
    public String aesDecode(@RequestParam String data) throws AesException {
        return AesHelper.decode(data);
    }

    @GetMapping("/rsa/sign")
    public String rsaSign(@RequestParam String data) throws RsaException {
        return RsaHelper.sign(data);
    }
}

密钥来源建议

⚠️ 不要把生产密钥提交到代码仓库或打包进 jar。建议通过环境变量、启动参数、配置中心或 KMS 注入：secret-key: ${TOOLS_AES_KEY}。当前 starter 不会在启动期拒绝不安全组合（如 DES/ECB、过短密钥）——请对照第 3 节安全边界自查配置。

6已知风险与限制

  1. ⚠️ DES 默认 ECB 且 IV 不生效。为保证既有密文可解默认模式未改动；建议优先使用 AES。
  2. ⚠️ JWT 硬编码默认签名串 TOOLS&DH$*001，未显式传参会使用它签发/校验令牌——生产必须显式传入。
  3. ⚠️ AesHelper 等基于静态配置（load）工作，一份 JVM 只有一份激活配置；多组密钥请使用 load(keyPair, name) 分组机制或显式传参重载。
  4. TODO：RSA 分段加密对超长明文的分段边界尚未做跨 Provider 互操作验证。

7常见问题

配置了 tools.encrypt.* 但调用 AesHelper.encode 报空指针 / 未初始化？

确认三件事：启动类上有 @EnableToolsEncrypt；对应算法 enabled: true；secret-key 已配置。三者缺一，该算法不会被装配。另注意 starter jar 内附的 application.yml 仅为示例，Spring Boot 不会加载依赖 jar 中的该文件。

加解密报 BadPaddingException 或解出乱码？

两端 algorithm-mode、密钥、IV 必须完全一致。常见错误：一端 AES/CBC/PKCS5Padding 另一端默认模式；IV 不是 16 字节；DES 两端密钥字节数不同。

AES 报 InvalidKeyException: Key length not 128/192/256 bits？

secret-key 是 Hex 字符串，解码后必须为 16/24/32 字节。检查是否误填了明文字符串或长度不足的 Hex。

DES 报 InvalidKeyException？

DES 密钥 UTF-8 编码后必须恰好 8 字节——例如 BeYourself（10 字节）就是非法密钥。

多环境密钥不同怎么办？

按 Spring Profile 放置不同 application-{profile}.yml，或使用分组 + load(keyPair, name) 运行期切换。

8测试与构建验证

模块测试为冒烟测试套件，覆盖 encrypt 全部公开方法（JUnit 4）。根 pom.xml 默认 skipTests=true（只编译不执行），执行测试：
